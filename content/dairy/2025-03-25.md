# 动态批处理优化方案

## 1. 数学建模改进

在现有的模型基础上，增加在线参数估计机制：

### 新增变量

### 改进后的优化问题
$$
\begin{aligned}
\min_{B} \ & \alpha (L_0 + \gamma_1 B + \gamma_2 Q) + \lambda \max\{0, L(B)-L_{\text{target}}\} \\
& - \beta r (1 - e^{-\delta B}) \\
\text{s.t.} \ & B \in \{1,2,...,B_{\text{max}}\} \\
& \theta \text{ 每 } T \text{ 秒更新一次}
\end{aligned}
$$

## 2. 实现方案选择

### 方案对比

| 方案 | 优点 | 缺点 | 适用场景 |
|------|------|------|---------|
| Prometheus指标暴露 | 1. 与现有监控体系集成度高<br>2. 可视化方便<br>3. 可复用现有告警机制 | 1. 引入额外网络开销<br>2. 数据采集有延迟<br>3. 需要额外配置 | 大规模生产环境 |
| Agent内部实时计算 | 1. 延迟低<br>2. 无网络依赖<br>3. 实现简单 | 1. 数据持久化困难<br>2. 调试不便<br>3. 内存占用高 | 开发测试环境或小规模部署 |

基于您的需求，我推荐**混合方案**：在Agent内部实现核心参数的实时计算，同时暴露关键指标给Prometheus用于监控和调试。

## 3. 代码实现

### 3.1 新增指标收集结构

```go
// 在batcher包中添加以下结构
type BatchMetrics struct {
    BatchSize        int
    QueueLength      int
    ProcessingTimeMs float64
    Timestamp        time.Time
}

type MetricsCollector struct {
    windowSize  int
    metricsRing []BatchMetrics
    pointer     int
    filled      bool
    mutex       sync.RWMutex
}

func NewMetricsCollector(windowSize int) *MetricsCollector {
    return &MetricsCollector{
        windowSize:  windowSize,
        metricsRing: make([]BatchMetrics, windowSize),
        pointer:     0,
        filled:      false,
    }
}

func (mc *MetricsCollector) AddMetric(metric BatchMetrics) {
    mc.mutex.Lock()
    defer mc.mutex.Unlock()
    
    mc.metricsRing[mc.pointer] = metric
    mc.pointer = (mc.pointer + 1) % mc.windowSize
    if !mc.filled && mc.pointer == 0 {
        mc.filled = true
    }
}

func (mc *MetricsCollector) GetRecentMetrics(n int) []BatchMetrics {
    mc.mutex.RLock()
    defer mc.mutex.RUnlock()
    
    if n <= 0 || n > mc.windowSize {
        n = mc.windowSize
    }
    if !mc.filled && n > mc.pointer {
        n = mc.pointer
    }
    
    result := make([]BatchMetrics, n)
    start := (mc.pointer - n + mc.windowSize) % mc.windowSize
    for i := 0; i < n; i++ {
        pos := (start + i) % mc.windowSize
        result[i] = mc.metricsRing[pos]
    }
    return result
}
```

### 3.2 参数估计实现

```go
// 在dynamic_adjuster.go中添加
func (handler *BatchHandler) estimateParameters() (gamma1, gamma2, delta float64, err error) {
    metrics := handler.metricsCollector.GetRecentMetrics(handler.metricsWindowSize)
    if len(metrics) < 10 { // 需要足够样本
        return gamma1, gamma2, delta, fmt.Errorf("not enough samples")
    }

    // 实现简单的线性回归
    var sumB, sumQ, sumL, sumB2, sumQ2, sumBQ, sumBL, sumQL float64
    n := float64(len(metrics))
    
    for _, m := range metrics {
        b := float64(m.BatchSize)
        q := float64(m.QueueLength)
        l := m.ProcessingTimeMs
        
        sumB += b
        sumQ += q
        sumL += l
        sumB2 += b * b
        sumQ2 += q * q
        sumBQ += b * q
        sumBL += b * l
        sumQL += q * l
    }
    
    // 解线性方程组: L = L0 + gamma1*B + gamma2*Q
    det := sumB2*sumQ2 - sumBQ*sumBQ
    if math.Abs(det) < 1e-10 {
        return gamma1, gamma2, delta, fmt.Errorf("singular matrix")
    }
    
    gamma1 = (sumQ2*sumBL - sumBQ*sumQL) / det
    gamma2 = (sumB2*sumQL - sumBQ*sumBL) / det
    
    // 简单估计delta (指数拟合)
    var sumBExp, sumLExp float64
    for _, m := range metrics {
        b := float64(m.BatchSize)
        sumBExp += b
        sumLExp += math.Log(1 - m.ProcessingTimeMs/(L0+gamma1*b+gamma2*float64(m.QueueLength)))
    }
    delta = -sumLExp / sumBExp
    
    return gamma1, gamma2, delta, nil
}
```

### 3.3 修改AdjustDynamicBatchSizeLoop

```go
func (handler *BatchHandler) AdjustDynamicBatchSizeLoop() {
    ticker := time.NewTicker(adjustment_interval)
    defer ticker.Stop()
    
    handler.metricsCollector = NewMetricsCollector(100) // 滑动窗口大小100
    handler.metricsWindowSize = 50 // 用于参数估计的样本数
    
    for {
        <-ticker.C
        
        // 1. 自动调整参数权重
        autoAdjustParamWeights(handler.log)
        
        // 2. 获取当前状态并记录指标
        Q := handler.getCurrentQueueLength()
        L0 = updateBaseLatency()
        r := updateArrivalRate()
        
        // 3. 更新参数估计
        gamma1, gamma2, delta, err := handler.estimateParameters()
        if err != nil {
            handler.log.Warnf("Parameter estimation failed: %v, using defaults", err)
            gamma1 = 0.5
            gamma2 = 1.0
            delta = 0.1
        }
        
        // 4. 搜索最优批处理大小
        bestB := 1
        bestF := F(1, Q, r, gamma1, gamma2, delta)
        for B := 2; B <= handler.MaxBatchSize; B++ {
            currentF := F(B, Q, r, gamma1, gamma2, delta)
            if currentF < bestF {
                bestF = currentF
                bestB = B
            }
        }
        
        handler.log.Infof("Dynamic adjust: Q=%.2f, L0=%.2f, r=%.2f, gamma1=%.3f, gamma2=%.3f, delta=%.3f, bestB=%d, bestF=%.2f",
            Q, L0, r, gamma1, gamma2, delta, bestB, bestF)
        handler.setDynamicBatchSize(bestB)
    }
}
```
