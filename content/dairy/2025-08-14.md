# main.py 训练过程工程优化深度分析与改进计划

经过对当前 `main.py` 训练过程的深度分析，我发现了多个可以优化的方面，从代码结构、训练效率、监控到可维护性等多个维度。以下是详细的优化计划和实现方案。

## 1. 当前代码问题分析

### 1.1 代码结构与可维护性问题
- **训练逻辑与主流程耦合**：训练循环、验证循环、日志记录等逻辑都混合在主函数中，难以维护和复用
- **硬编码参数**：许多参数直接写在代码中，缺乏灵活性
- **缺乏配置化**：没有统一的配置管理，每次修改参数都需要改动代码
- **异常处理不足**：缺乏对训练过程中可能出现的异常的处理

### 1.2 训练效率问题
- **数据加载效率**：当前使用单线程数据加载，没有充分利用多核CPU
- **内存使用**：没有使用梯度累积，对于大模型或大batch size可能内存不足
- **计算效率**：没有使用混合精度训练，浪费了GPU的计算能力
- **学习率调度**：缺乏灵活的学习率调度策略

### 1.3 监控与调试问题
- **监控指标单一**：只记录损失值，缺乏对模型内部状态的监控
- **TensorBoard使用不充分**：没有充分利用TensorBoard的可视化能力
- **日志记录不完善**：缺乏详细的训练过程记录
- **模型状态保存不完整**：只保存模型权重，没有保存优化器状态等

### 1.4 可复现性与扩展性问题
- **随机种子未设置**：每次训练结果可能不同
- **环境信息未记录**：难以复现实验环境
- **缺乏模块化设计**：难以扩展新的训练策略或评估指标

## 2. 优化方案与实现

### 2.1 代码结构重构

#### 2.1.1 模块化设计
将训练过程拆分为多个独立模块，提高代码复用性和可维护性。

```python
# train.py - 训练模块
class Trainer:
    def __init__(self, model, optimizer, scheduler, criterion, args):
        self.model = model
        self.optimizer = optimizer
        self.scheduler = scheduler
        self.criterion = criterion
        self.args = args
        self.scaler = torch.cuda.amp.GradScaler(enabled=args.amp)
        self.global_step = 0
        self.best_metric = float('-inf')
        
    def train_epoch(self, train_loader, epoch):
        self.model.train()
        total_loss = 0.0
        progress_bar = tqdm(train_loader, desc=f"Epoch {epoch}")
        
        for batch_idx, batch in enumerate(progress_bar):
            loss = self._train_step(batch)
            total_loss += loss.item()
            
            # 更新进度条
            progress_bar.set_postfix({
                'loss': loss.item(),
                'lr': self.optimizer.param_groups[0]['lr']
            })
            
            # 记录到TensorBoard
            if batch_idx % self.args.log_interval == 0:
                self._log_metrics(loss, 'train')
            
            self.global_step += 1
        
        return total_loss / len(train_loader)
    
    def _train_step(self, batch):
        self.optimizer.zero_grad()
        
        with torch.cuda.amp.autocast(enabled=self.args.amp):
            seq_embs, pos_embs, neg_embs, loss_mask = self.model(*batch)
            loss = self.criterion(seq_embs, pos_embs, neg_embs, loss_mask)
            loss = loss / self.args.accumulation_steps
        
        self.scaler.scale(loss).backward()
        
        if (self.global_step + 1) % self.args.accumulation_steps == 0:
            self.scaler.unscale_(self.optimizer)
            torch.nn.utils.clip_grad_norm_(self.model.parameters(), self.args.max_grad_norm)
            self.scaler.step(self.optimizer)
            self.scaler.update()
            
            if self.scheduler is not None:
                self.scheduler.step()
        
        return loss
    
    def validate(self, valid_loader):
        self.model.eval()
        total_loss = 0.0
        
        with torch.no_grad():
            for batch in tqdm(valid_loader, desc="Validation"):
                seq_embs, pos_embs, neg_embs, loss_mask = self.model(*batch)
                loss = self.criterion(seq_embs, pos_embs, neg_embs, loss_mask)
                total_loss += loss.item()
        
        avg_loss = total_loss / len(valid_loader)
        self._log_metrics(avg_loss, 'valid')
        return avg_loss
    
    def _log_metrics(self, loss, prefix):
        # 记录损失
        self.writer.add_scalar(f"{prefix}/loss", loss, self.global_step)
        
        # 记录学习率
        self.writer.add_scalar(f"{prefix}/lr", self.optimizer.param_groups[0]['lr'], self.global_step)
        
        # 记录梯度范数
        if prefix == 'train':
            total_norm = 0
            for p in self.model.parameters():
                if p.grad is not None:
                    param_norm = p.grad.data.norm(2)
                    total_norm += param_norm.item() ** 2
            total_norm = total_norm ** 0.5
            self.writer.add_scalar(f"{prefix}/grad_norm", total_norm, self.global_step)
    
    def save_checkpoint(self, epoch, metric, is_best=False):
        checkpoint = {
            'epoch': epoch,
            'global_step': self.global_step,
            'state_dict': self.model.state_dict(),
            'optimizer': self.optimizer.state_dict(),
            'scheduler': self.scheduler.state_dict() if self.scheduler else None,
            'best_metric': self.best_metric,
            'args': self.args
        }
        
        # 创建检查点目录
        checkpoint_dir = Path(self.args.checkpoint_dir)
        checkpoint_dir.mkdir(parents=True, exist_ok=True)
        
        # 保存最新检查点
        torch.save(checkpoint, checkpoint_dir / 'latest_checkpoint.pt')
        
        # 保存最佳检查点
        if is_best:
            torch.save(checkpoint, checkpoint_dir / 'best_checkpoint.pt')
        
        # 保存带时间戳的检查点
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        checkpoint_path = checkpoint_dir / f'checkpoint_epoch{epoch}_{timestamp}.pt'
        torch.save(checkpoint, checkpoint_path)
        
        return checkpoint_path
```

```python
                'pin_memory': True,
                'prefetch_factor': 2
            }
        }
    
    def _update_from_file(self, config_path):
        with open(config_path, 'r') as f:
            file_config = yaml.safe_load(f)
        
        def update_dict(d, u):
            for k, v in u.items():
                if isinstance(v, dict):
                    d[k] = update_dict(d.get(k, {}), v)
                else:
                    d[k] = v
            return d
        
        self._config = update_dict(self._config, file_config)
    
    def __getattr__(self, name):
        if name in self._config:
            return self._config[name]
        raise AttributeError(f"'{type(self).__name__}' object has no attribute '{name}'")
    
    def to_namespace(self):
        """将配置转换为argparse.Namespace对象"""
        def dict_to_namespace(d):
            if isinstance(d, dict):
                namespace = argparse.Namespace()
                for key, value in d.items():
                    setattr(namespace, key, dict_to_namespace(value))
                return namespace
            return d
        
        return dict_to_namespace(self._config)
```

```python
# lr_scheduler.py
class LRSchedulerFactory:
    @staticmethod
    def create_scheduler(optimizer, args):
        name = args.optimization.lr_scheduler.lower()
        
        if name == 'step':
            return torch.optim.lr_scheduler.StepLR(
                optimizer,
                step_size=args.optimization.lr_decay_steps,
                gamma=args.optimization.lr_decay_rate
            )
        elif name == 'cosine':
            return torch.optim.lr_scheduler.CosineAnnealingLR(
                optimizer,
                T_max=args.training.num_epochs,
                eta_min=1e-6
            )
        elif name == 'cosine_warm_restarts':
            return torch.optim.lr_scheduler.CosineAnnealingWarmRestarts(
                optimizer,
                T_0=10,
                T_mult=2,
                eta_min=1e-6
            )
        elif name == 'linear':
            return torch.optim.lr_scheduler.LinearLR(
                optimizer,
                start_factor=1.0,
                end_factor=0.01,
                total_iters=args.training.num_epochs
            )
        elif name == 'one_cycle':
            return torch.optim.lr_scheduler.OneCycleLR(
                optimizer,
                max_lr=args.training.lr,
                epochs=args.training.num_epochs,
                steps_per_epoch=len(train_loader),
                pct_start=0.3,
                anneal_strategy='cos'
            )
        else:
            return None
```

```python
# tensorboard_utils.py
class TensorBoardLogger:
    def __init__(self, log_dir):
        self.writer = SummaryWriter(log_dir)
        self.step = 0
    
    def log_metrics(self, metrics, prefix):
        for name, value in metrics.items():
            self.writer.add_scalar(f"{prefix}/{name}", value, self.step)
    
    def log_model_parameters(self, model):
        for name, param in model.named_parameters():
            if param.requires_grad:
                self.writer.add_histogram(f"parameters/{name}", param.data, self.step)
                if param.grad is not None:
                    self.writer.add_histogram(f"gradients/{name}", param.grad.data, self.step)
    
    def log_learning_rate(self, optimizer):
        for i, param_group in enumerate(optimizer.param_groups):
            self.writer.add_scalar(f"learning_rate/group_{i}", param_group['lr'], self.step)
    
    def log_embeddings(self, embeddings, metadata=None):
        self.writer.add_embedding(embeddings, metadata=metadata, global_step=self.step)
    
    def log_graph(self, model, input_sample):
        self.writer.add_graph(model, input_sample)
    
    def increment_step(self):
        self.step += 1
    
    def close(self):
        self.writer.close()
```

```python
# training_monitor.py
class TrainingMonitor:
    def __init__(self, patience=5, min_delta=1e-4):
        self.patience = patience
        self.min_delta = min_delta
        self.best_metric = float('-inf')
        self.counter = 0
        self.should_stop = False
    
    def update(self, current_metric):
        if current_metric > self.best_metric + self.min_delta:
            self.best_metric = current_metric
            self.counter = 0
            return True  # 改进了
        else:
            self.counter += 1
            if self.counter >= self.patience:
                self.should_stop = True
            return False  # 没有改进
    
    def reset(self):
        self.best_metric = float('-inf')
        self.counter = 0
        self.should_stop = False
```

```python
# metrics.py
class MetricCalculator:
    @staticmethod
    def hit_ratio(logits, labels, k=10):
        """计算Hit Ratio@K"""
        _, indices = torch.topk(logits, k)
        labels = labels.view(-1, 1)
        hits = (indices == labels).nonzero()
        return len(hits) / logits.size(0)
    
    @staticmethod
    def ndcg(logits, labels, k=10):
        """计算NDCG@K"""
        _, indices = torch.topk(logits, k)
        labels = labels.view(-1, 1)
        hits = (indices == labels).nonzero()
        if len(hits) == 0:
            return 0.0
        
        ranks = hits[:, -1] + 1
        dcg = torch.sum(1 / torch.log2(ranks.float() + 1))
        idcg = torch.sum(1 / torch.log2(torch.arange(1, min(k, logits.size(0)) + 1).float() + 1))
        return (dcg / idcg).item()
    
    @staticmethod
    def mrr(logits, labels):
        """计算Mean Reciprocal Rank"""
        _, indices = torch.sort(logits, descending=True)
        labels = labels.view(-1, 1)
        ranks = (indices == labels).nonzero()[:, -1] + 1
        return torch.mean(1.0 / ranks.float()).item()
```

```python

def setup_logging(log_dir):
    """设置日志记录"""
    log_dir = Path(log_dir)
    log_dir.mkdir(parents=True, exist_ok=True)
    
    log_file = log_dir / f'train_{datetime.now().strftime("%Y%m%d_%H%M%S")}.log'
    
    logging.basicConfig(
        level=logging.INFO,
        format='%(asctime)s - %(levelname)s - %(message)s',
        handlers=[
            logging.FileHandler(log_file),
            logging.StreamHandler()
        ]
    )
    
    return log_file

def main():
    # 解析命令行参数
    parser = argparse.ArgumentParser(description='Transformer-based Recommendation System')
    parser.add_argument('--config', type=str, default=None, help='Path to config file')
    parser.add_argument('--checkpoint', type=str, default=None, help='Path to checkpoint')
    parser.add_argument('--mode', type=str, default='train', choices=['train', 'eval'], help='Run mode')
    args = parser.parse_args()
    
    # 加载配置
    config = Config(args.config)
    args = config.to_namespace()
    
    # 设置随机种子
    set_seed(args.training.seed)
    
    # 设置日志

    logging.info(json.dumps(vars(args), indent=2))

    with open(log_file, 'a') as f:

    valid_size = len(dataset) - train_size

        dataset, [train_size, valid_size],
        generator=torch.Generator().manual_seed(args.training.seed)
    )
    
    # 创建数据加载器
    train_loader = DataLoaderFactory.create_train_loader(train_dataset, args)
    valid_loader = DataLoaderFactory.create_valid_loader(valid_dataset, args)

        dataset.usernum, 
        dataset.itemnum, 
        dataset.feat_statistics, 
        dataset.feature_types, 
        args

        model.parameters(),
        lr=args.training.lr,
        betas=(0.9, 0.98),
        weight_decay=args.optimization.weight_decay
    )
    
    # 创建学习率调度器
    scheduler = LRSchedulerFactory.create_scheduler(optimizer, args)
    
    # 创建损失函数
    criterion = model.compute_infonce_loss
    
    # 创建TensorBoard日志记录器
    tb_logger = TensorBoardLogger(args.tensorboard_dir)
    
    # 创建训练监控器
    monitor = TrainingMonitor(patience=args.patience, min_delta=args.min_delta)
    
    # 创建训练器
    trainer = Trainer(
        model=model,
        optimizer=optimizer,
        scheduler=scheduler,
        criterion=criterion,
        args=args,
        tb_logger=tb_logger
    )
    
    # 加载检查点
    start_epoch = 0
    if args.checkpoint:
        checkpoint = torch.load(args.checkpoint, map_location=args.system.device)
        model.load_state_dict(checkpoint['state_dict'])
        optimizer.load_state_dict(checkpoint['optimizer'])
        if scheduler and 'scheduler' in checkpoint and checkpoint['scheduler']:
            scheduler.load_state_dict(checkpoint['scheduler'])
        start_epoch = checkpoint['epoch'] + 1
        trainer.global_step = checkpoint['global_step']
        monitor.best_metric = checkpoint['best_metric']
        logging.info(f"Loaded checkpoint from {args.checkpoint}, starting at epoch {start_epoch}")
    
    # 记录模型图
    if args.mode == 'train':
        try:

            tb_logger.log_graph(model, sample_batch[:5])  # 只使用部分样本避免内存问题
        except Exception as e:
            logging.warning(f"Failed to log model graph: {e}")
    
    # 训练循环
    for epoch in range(start_epoch, args.training.num_epochs):
        logging.info(f"Starting epoch {epoch}/{args.training.num_epochs}")
        
        # 训练一个epoch
        train_loss = trainer.train_epoch(train_loader, epoch)
        logging.info(f"Epoch {epoch} train loss: {train_loss:.4f}")
        
        # 验证
        if epoch % args.logging.eval_interval == 0:
            valid_loss = trainer.validate(valid_loader)
            logging.info(f"Epoch {epoch} valid loss: {valid_loss:.4f}")
            
            # 计算评估指标
            model.eval()
            hr_list, ndcg_list, mrr_list = [], [], []
            
            with torch.no_grad():
                for batch in tqdm(valid_loader, desc="Evaluating"):
                    seq, pos, neg, token_type, next_token_type, next_action_type, seq_feat, pos_feat, neg_feat = batch
                    seq = seq.to(args.system.device)
                    pos = pos.to(args.system.device)
                    
                    # 获取用户表示
                    user_emb = model.predict(seq, seq_feat, token_type)
                    
                    # 获取正样本表示
                    pos_emb = model.feat2emb(pos, pos_feat, include_user=False)
                    
                    # 计算相似度
                    logits = torch.matmul(user_emb, pos_emb.transpose(-1, -2))
                    
                    # 计算指标
                    for i in range(logits.size(0)):
                        labels = torch.arange(logits.size(1)).to(args.system.device)
                        hr_list.append(MetricCalculator.hit_ratio(logits[i], labels[i]))
                        ndcg_list.append(MetricCalculator.ndcg(logits[i], labels[i]))
                        mrr_list.append(MetricCalculator.mrr(logits[i], labels[i]))
            
            avg_hr = np.mean(hr_list)
            avg_ndcg = np.mean(ndcg_list)
            avg_mrr = np.mean(mrr_list)
            
            logging.info(f"Epoch {epoch} HR@10: {avg_hr:.4f}, NDCG@10: {avg_ndcg:.4f}, MRR: {avg_mrr:.4f}")
            
            # 记录到TensorBoard
            tb_logger.log_metrics({
                'hr@10': avg_hr,
                'ndcg@10': avg_ndcg,
                'mrr': avg_mrr
            }, 'valid')
            
            # 更新监控器
            is_best = monitor.update(avg_ndcg)
            
            # 保存检查点
            if epoch % args.logging.save_interval == 0 or is_best:
                checkpoint_path = trainer.save_checkpoint(
                    epoch=epoch,
                    metric=avg_ndcg,
                    is_best=is_best
                )
                logging.info(f"Saved checkpoint to {checkpoint_path}")
            
            # 早停检查
            if monitor.should_stop:
                logging.info(f"Early stopping triggered after {epoch} epochs")
                break
    
    # 关闭TensorBoard
    tb_logger.close()
    logging.info("Training completed!")

if __name__ == '__main__':
    main()
```
